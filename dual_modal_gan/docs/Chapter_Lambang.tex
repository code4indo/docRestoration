% ============================================================
% DATA LAMBANG DAN SIMBOL (KATEGORIS)
% ============================================================

\multicolumn{2}{l}{\textbf{Huruf Yunani (Parameter Model)}} \\

$\alpha$ & Koefisien bobot untuk \textit{loss function}, tingkat signifikansi statistik \\
$\beta$ & Koefisien bobot untuk \textit{Binary Cross-Entropy loss} \\
$\lambda$ & Koefisien bobot untuk CTC \textit{loss}, parameter regularisasi \\
$\theta$ & Parameter model \textit{neural network} \\
$\sigma$ & Deviasi standar \\
$\mu$ & Rata-rata (\textit{mean}) \\
$\epsilon$ & \textit{Token blank}/kosong dalam CTC, konstanta kecil untuk stabilitas numerik \\
$\Delta$ & Delta (selisih/perubahan nilai) \\
$\pi$ & \textit{Alignment path} dalam CTC \\
$\pi_t$ & \textit{Token} pada \textit{timestep} $t$ dalam \textit{alignment path} \\[1ex]

\multicolumn{2}{l}{\textbf{Operator Matematis}} \\

$\nabla$ atau $\partial$ & Operator gradien atau turunan parsial \\
$\sum$ & Operator penjumlahan \\
$\prod$ & Operator perkalian \\
$\int$ & Operator integral \\
argmax & Argumen yang memaksimalkan fungsi \\
$\sim$ & Terdistribusi menurut \\
$\to$ & Menuju atau konvergen ke \\[1ex]

\multicolumn{2}{l}{\textbf{Fungsi \textit{Loss} dan Kerugian}} \\

$\mathcal{L}$ & Fungsi \textit{loss} (kerugian) \\
$\mathcal{L}_{total}$ & Total \textit{loss function} \\
$\mathcal{L}_{adversarial}$ atau $\mathcal{L}_{adv}$ & \textit{Adversarial loss} \\
$\mathcal{L}_{reconstruction}$ & \textit{Reconstruction loss} \\
$\mathcal{L}_{L1}$ & L1 \textit{loss} (\textit{Mean Absolute Error}) \\
$\mathcal{L}_{L2}$ & L2 \textit{loss} (\textit{Mean Squared Error}) \\
$\mathcal{L}_{CTC}$ & CTC (\textit{Connectionist Temporal Classification}) \textit{loss} \\
$\mathcal{L}_{perceptual}$ & \textit{Perceptual loss} \\
$\mathcal{L}_{pixel}$ & \textit{Pixel-wise loss} \\
$\mathcal{L}_{BCE}$ & \textit{Binary Cross-Entropy loss} \\
$\mathcal{L}_{GAN}$ & GAN \textit{loss} \\
$\mathcal{L}_{cycle}$ & \textit{Cycle consistency loss} \\[1ex]

\multicolumn{2}{l}{\textbf{Variabel Model dan Komponen GAN}} \\

$G$ & \textit{Generator} dalam GAN \\
$D$ & \textit{Discriminator} dalam GAN \\
$G(z)$ & \textit{Output} dari \textit{generator} dengan \textit{input} $z$ \\
$D(x)$ & \textit{Output} dari \textit{discriminator} dengan \textit{input} $x$ \\
$G_{A \to B}$ & \textit{Generator} dari domain $A$ ke $B$ \\
$G_{B \to A}$ & \textit{Generator} dari domain $B$ ke $A$ \\
$V(D, G)$ & Fungsi nilai (\textit{value function}) dalam GAN \\
$x$ & Sampel data asli, \textit{input} citra \\
$y$ & Label atau \textit{ground truth}, kondisi dalam \textit{conditional GAN} \\
$z$ & \textit{Noise vector} atau representasi laten \\[1ex]

\multicolumn{2}{l}{\textbf{Distribusi Probabilitas}} \\

$\mathbb{E}$ & Ekspektasi atau nilai harapan \\
$p_{data}$ & Distribusi data asli \\
$p_z$ & Distribusi \textit{prior} (\textit{noise}) \\
$p(y|x)$ & Probabilitas bersyarat \textit{output} $y$ \textit{given input} $x$ \\
$p_t(\pi_t|x)$ & Probabilitas \textit{token} pada \textit{timestep} $t$ \\[1ex]

\multicolumn{2}{l}{\textbf{Fungsi dan Transformasi CTC}} \\

$\mathcal{B}$ & Fungsi penciutan (\textit{collapse function}) dalam CTC \\
$\mathcal{B}^{-1}$ & Fungsi invers penciutan dalam CTC \\
$T$ & Jumlah \textit{timestep} atau panjang \textit{sequence} \\[1ex]

\multicolumn{2}{l}{\textbf{Metrik Jarak dan Norma}} \\

$|x - G(y)|_1$ & L1 \textit{distance} antara \textit{ground truth} dan \textit{generated image} \\
$|x - G(y)|_2$ & L2 \textit{distance} antara \textit{ground truth} dan \textit{generated image} \\
$N \times N$ & Ukuran \textit{patch} dalam PatchGAN \\[1ex]

\multicolumn{2}{l}{\textbf{Statistik dan Analisis}} \\

$d$ & Cohen's $d$ (\textit{effect size}) \\
$n$ & Jumlah sampel \\
$p$ & \textit{p-value} (nilai probabilitas statistik) \\
$r$ & Koefisien korelasi \\
$R^2$ & Koefisien determinasi \\[1ex]

\multicolumn{2}{l}{\textbf{Notasi Kompleksitas Komputasi}} \\

$O(n)$ & Notasi Big-O untuk kompleksitas linear \\
$O(n^2)$ & Notasi Big-O untuk kompleksitas kuadratik \\
